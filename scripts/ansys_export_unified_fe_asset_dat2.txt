! ============================================================================
! Export one unified ANSYS modal transfer asset from the current result set.
!
! This file is tailored to:
!   齿圈模态\2\2\1.dat
! Named selections present in that model:
!   SENSOR_0              9-node measured patch
!   RING_MESHING_SURFACE  inner-ring meshing surface
!
! The script reuses the solved RST; it does not solve the FE model again.
! SENSOR_120 and SENSOR_240 are generated by CLOCKWISE rotation of SENSOR_0
! coordinates about the ring centre and selecting the nearest full-mesh nodes.
! Local sensor labels are clockwise offsets; global atan2 is counterclockwise.
! Mapping errors are exported for the Python asset builder to reject an
! incomplete patch or a node farther than MAX_MAPPING_ERROR_MM. FINISH alone
! does not guarantee that the remainder of this macro is skipped. SENSOR_0 is on the
! outer structure, not on RING_MESHING_SURFACE, so search the full FE mesh.
!
! Outputs (Workbench user_files directory):
!   modal_summary.csv
!   mode_shapes_ring_teeth.csv
!   mode_shapes_sensor.csv
!   sensor_mapping.csv
!   auto_sensor_check.csv
!   UNIFIED_FE_ASSET_EXPORT_FINISHED.txt
!
! Important: the model is in N-mm-MPa. Coordinates and modal displacements
! are explicitly converted to metres in the CSV files.
! ============================================================================

FINISH
/POST1
CSYS,0
RSYS,0
*AFUN,DEG
ALLSEL,ALL

! Number of result sets. This solve contains 60 Block-Lanczos modes.
*GET,NMODES,ACTIVE,0,SET,NSET
*IF,NMODES,LE,0,THEN
  /COM,ERROR: no modal result sets are active; CSV export will be rejected.
  NMODES=0
*ENDIF
*IF,NMODES,GT,60,THEN
  NMODES=60
*ENDIF

! Ring centre from the meshing surface component.
CMSEL,S,RING_MESHING_SURFACE,NODE
*GET,XRMIN,NODE,0,MNLOC,X
*GET,XRMAX,NODE,0,MXLOC,X
*GET,YRMIN,NODE,0,MNLOC,Y
*GET,YRMAX,NODE,0,MXLOC,Y
XC=(XRMIN+XRMAX)/2
YC=(YRMIN+YRMAX)/2

! Read SENSOR_0 in deterministic node order.
ALLSEL,ALL
CMSEL,S,SENSOR_0,NODE
*GET,NS0,NODE,0,COUNT
*IF,NS0,LE,0,THEN
  /COM,ERROR: SENSOR_0 is empty.
  *MSG,,'SENSOR_0 is empty.'
  FINISH
*ENDIF
*DIM,S0_ID,ARRAY,NS0
*DIM,S0_X,ARRAY,NS0
*DIM,S0_Y,ARRAY,NS0
*DIM,S0_Z,ARRAY,NS0
*GET,NID,NODE,0,NUM,MIN
*DO,II,1,NS0
  S0_ID(II)=NID
  *GET,S0_X(II),NODE,NID,LOC,X
  *GET,S0_Y(II),NODE,NID,LOC,Y
  *GET,S0_Z(II),NODE,NID,LOC,Z
  *GET,NID,NODE,NID,NXTH
*ENDDO

! Generate the two circumferential patches from SENSOR_0.
*DIM,S120_ID,ARRAY,NS0
*DIM,S240_ID,ARRAY,NS0
*DIM,T120_X,ARRAY,NS0
*DIM,T120_Y,ARRAY,NS0
*DIM,T120_Z,ARRAY,NS0
*DIM,T240_X,ARRAY,NS0
*DIM,T240_Y,ARRAY,NS0
*DIM,T240_Z,ARRAY,NS0
*DIM,DIST120,ARRAY,NS0
*DIM,DIST240,ARRAY,NS0
SUMD120=0
SUMD240=0
MAXD120=0
MAXD240=0
MAX_MAPPING_ERROR_MM=2.0
ALLSEL,ALL
NSEL,ALL
*DO,II,1,NS0
  DX=S0_X(II)-XC
  DY=S0_Y(II)-YC

  T120_X(II)=XC+DX*COS(120)+DY*SIN(120)
  T120_Y(II)=YC-DX*SIN(120)+DY*COS(120)
  T120_Z(II)=S0_Z(II)
  S120_ID(II)=NODE(T120_X(II),T120_Y(II),T120_Z(II))
  *GET,XA120,NODE,S120_ID(II),LOC,X
  *GET,YA120,NODE,S120_ID(II),LOC,Y
  *GET,ZA120,NODE,S120_ID(II),LOC,Z
  DIST120(II)=SQRT((XA120-T120_X(II))**2+(YA120-T120_Y(II))**2+(ZA120-T120_Z(II))**2)
  SUMD120=SUMD120+DIST120(II)
  *IF,DIST120(II),GT,MAXD120,THEN
    MAXD120=DIST120(II)
  *ENDIF

  T240_X(II)=XC+DX*COS(240)+DY*SIN(240)
  T240_Y(II)=YC-DX*SIN(240)+DY*COS(240)
  T240_Z(II)=S0_Z(II)
  S240_ID(II)=NODE(T240_X(II),T240_Y(II),T240_Z(II))
  *GET,XA240,NODE,S240_ID(II),LOC,X
  *GET,YA240,NODE,S240_ID(II),LOC,Y
  *GET,ZA240,NODE,S240_ID(II),LOC,Z
  DIST240(II)=SQRT((XA240-T240_X(II))**2+(YA240-T240_Y(II))**2+(ZA240-T240_Z(II))**2)
  SUMD240=SUMD240+DIST240(II)
  *IF,DIST240(II),GT,MAXD240,THEN
    MAXD240=DIST240(II)
  *ENDIF
*ENDDO
MEAND120=SUMD120/NS0
MEAND240=SUMD240/NS0

! Build the generated sensor components and verify unique counts.
ALLSEL,ALL
NSEL,NONE
*DO,II,1,NS0
  NSEL,A,NODE,,S120_ID(II)
*ENDDO
CM,AUTO_SENSOR_120,NODE
*GET,NS120,NODE,0,COUNT

ALLSEL,ALL
NSEL,NONE
*DO,II,1,NS0
  NSEL,A,NODE,,S240_ID(II)
*ENDDO
CM,AUTO_SENSOR_240,NODE
*GET,NS240,NODE,0,COUNT
ALLSEL,ALL

*IF,NS120,NE,NS0,THEN
  /COM,ERROR: SENSOR_120 mapping is not one-to-one.
  FINISH
*ENDIF
*IF,NS240,NE,NS0,THEN
  /COM,ERROR: SENSOR_240 mapping is not one-to-one.
  FINISH
*ENDIF
*IF,MAXD120,GT,MAX_MAPPING_ERROR_MM,THEN
  /COM,ERROR: SENSOR_120 mapping error exceeds MAX_MAPPING_ERROR_MM.
  FINISH
*ENDIF
*IF,MAXD240,GT,MAX_MAPPING_ERROR_MM,THEN
  /COM,ERROR: SENSOR_240 mapping error exceeds MAX_MAPPING_ERROR_MM.
  FINISH
*ENDIF

! Mapping and node-count QA.
*CFOPEN,%_WB_USERFILES_DIR_MODAL(1)%auto_sensor_check,csv
*VWRITE,'item','value'
(A,',',A)
*VWRITE,'ring_cx',XC
(A,',',E20.12)
*VWRITE,'ring_cy',YC
(A,',',E20.12)
*VWRITE,'n_s0',NS0
(A,',',I12)
*VWRITE,'n_s120',NS120
(A,',',I12)
*VWRITE,'n_s240',NS240
(A,',',I12)
*VWRITE,'mean120',MEAND120
(A,',',E20.12)
*VWRITE,'max120',MAXD120
(A,',',E20.12)
*VWRITE,'mean240',MEAND240
(A,',',E20.12)
*VWRITE,'max240',MAXD240
(A,',',E20.12)
*VWRITE,'unit','mm_to_m'
(A,',',A)
*CFCLOS

*CFOPEN,%_WB_USERFILES_DIR_MODAL(1)%sensor_mapping,csv
*VWRITE,'src_node','angle','dst_node','target_x','target_y','target_z','mapped_x','mapped_y','mapped_z','err_mm'
(A,',',A,',',A,',',A,',',A,',',A,',',A,',',A,',',A,',',A)
*DO,II,1,NS0
  *GET,XA120,NODE,S120_ID(II),LOC,X
  *GET,YA120,NODE,S120_ID(II),LOC,Y
  *GET,ZA120,NODE,S120_ID(II),LOC,Z
  *GET,XA240,NODE,S240_ID(II),LOC,X
  *GET,YA240,NODE,S240_ID(II),LOC,Y
  *GET,ZA240,NODE,S240_ID(II),LOC,Z
  *VWRITE,S0_ID(II),120,S120_ID(II),T120_X(II),T120_Y(II),T120_Z(II),XA120,YA120,ZA120,DIST120(II)
(I8,',',I4,',',I8,',',E12.5,',',E12.5,',',E12.5,',',E12.5,',',E12.5,',',E12.5,',',E12.5)
  *VWRITE,S0_ID(II),240,S240_ID(II),T240_X(II),T240_Y(II),T240_Z(II),XA240,YA240,ZA240,DIST240(II)
(I8,',',I4,',',I8,',',E12.5,',',E12.5,',',E12.5,',',E12.5,',',E12.5,',',E12.5,',',E12.5)
*ENDDO
*CFCLOS

! Frequency table. MODE,FREQ is used instead of ACTIVE,SET,FREQ so that
! the exported modal frequencies are nonzero and auditable.
*CFOPEN,%_WB_USERFILES_DIR_MODAL(1)%modal_summary,csv
*VWRITE,'mode_id','freq_Hz'
(A,',',A)
*DO,IMODE,1,NMODES
  *GET,FREQVAL,MODE,IMODE,FREQ
  *VWRITE,IMODE,FREQVAL
(I8,',',E20.12)
*ENDDO
*CFCLOS

! Export all nodes of the RING_MESHING_SURFACE. Coordinates and modal
! displacements are converted from mm to m for the Python/MATLAB asset loader.
*CFOPEN,%_WB_USERFILES_DIR_MODAL(1)%mode_shapes_ring_teeth,csv
*VWRITE,'mode_id','node_id','x_m','y_m','z_m','ux_m','uy_m','uz_m'
(A,',',A,',',A,',',A,',',A,',',A,',',A,',',A)
*DO,IMODE,1,NMODES
  SET,1,IMODE
  ALLSEL,ALL
  CMSEL,S,RING_MESHING_SURFACE,NODE
  *GET,NNODE,NODE,0,COUNT
  *GET,NID,NODE,0,NUM,MIN
  *DO,II,1,NNODE
    *GET,XVAL,NODE,NID,LOC,X
    *GET,YVAL,NODE,NID,LOC,Y
    *GET,ZVAL,NODE,NID,LOC,Z
    *GET,UXVAL,NODE,NID,U,X
    *GET,UYVAL,NODE,NID,U,Y
    *GET,UZVAL,NODE,NID,U,Z
    *VWRITE,IMODE,NID,XVAL*1E-3,YVAL*1E-3,ZVAL*1E-3,UXVAL*1E-3,UYVAL*1E-3,UZVAL*1E-3
(I6,',',I9,',',E14.7,',',E14.7,',',E14.7,',',E14.7,',',E14.7,',',E14.7)
    *GET,NID,NODE,NID,NXTH
  *ENDDO
*ENDDO
*CFCLOS

! Export the three sensor patches from exactly the same result sets.
*CFOPEN,%_WB_USERFILES_DIR_MODAL(1)%mode_shapes_sensor,csv
*VWRITE,'mode_id','angle','node_id','x_m','y_m','z_m','ux_m','uy_m','uz_m'
(A,',',A,',',A,',',A,',',A,',',A,',',A,',',A,',',A)
*DO,IMODE,1,NMODES
  SET,1,IMODE
  ALLSEL,ALL
  CMSEL,S,SENSOR_0,NODE
  *GET,NNODE,NODE,0,COUNT
  *GET,NID,NODE,0,NUM,MIN
  *DO,II,1,NNODE
    *GET,XVAL,NODE,NID,LOC,X
    *GET,YVAL,NODE,NID,LOC,Y
    *GET,ZVAL,NODE,NID,LOC,Z
    *GET,UXVAL,NODE,NID,U,X
    *GET,UYVAL,NODE,NID,U,Y
    *GET,UZVAL,NODE,NID,U,Z
    *VWRITE,IMODE,0,NID,XVAL*1E-3,YVAL*1E-3,ZVAL*1E-3,UXVAL*1E-3,UYVAL*1E-3,UZVAL*1E-3
(I6,',',I4,',',I9,',',E14.7,',',E14.7,',',E14.7,',',E14.7,',',E14.7,',',E14.7)
    *GET,NID,NODE,NID,NXTH
  *ENDDO
  ALLSEL,ALL
  CMSEL,S,AUTO_SENSOR_120,NODE
  *GET,NNODE,NODE,0,COUNT
  *GET,NID,NODE,0,NUM,MIN
  *DO,II,1,NNODE
    *GET,XVAL,NODE,NID,LOC,X
    *GET,YVAL,NODE,NID,LOC,Y
    *GET,ZVAL,NODE,NID,LOC,Z
    *GET,UXVAL,NODE,NID,U,X
    *GET,UYVAL,NODE,NID,U,Y
    *GET,UZVAL,NODE,NID,U,Z
    *VWRITE,IMODE,120,NID,XVAL*1E-3,YVAL*1E-3,ZVAL*1E-3,UXVAL*1E-3,UYVAL*1E-3,UZVAL*1E-3
(I6,',',I4,',',I9,',',E14.7,',',E14.7,',',E14.7,',',E14.7,',',E14.7,',',E14.7)
    *GET,NID,NODE,NID,NXTH
  *ENDDO
  ALLSEL,ALL
  CMSEL,S,AUTO_SENSOR_240,NODE
  *GET,NNODE,NODE,0,COUNT
  *GET,NID,NODE,0,NUM,MIN
  *DO,II,1,NNODE
    *GET,XVAL,NODE,NID,LOC,X
    *GET,YVAL,NODE,NID,LOC,Y
    *GET,ZVAL,NODE,NID,LOC,Z
    *GET,UXVAL,NODE,NID,U,X
    *GET,UYVAL,NODE,NID,U,Y
    *GET,UZVAL,NODE,NID,U,Z
    *VWRITE,IMODE,240,NID,XVAL*1E-3,YVAL*1E-3,ZVAL*1E-3,UXVAL*1E-3,UYVAL*1E-3,UZVAL*1E-3
(I6,',',I4,',',I9,',',E14.7,',',E14.7,',',E14.7,',',E14.7,',',E14.7,',',E14.7)
    *GET,NID,NODE,NID,NXTH
  *ENDDO
*ENDDO
*CFCLOS

*CFOPEN,%_WB_USERFILES_DIR_MODAL(1)%UNIFIED_FE_ASSET_EXPORT_FINISHED,txt
*VWRITE,'OK: unified FE modal asset exported from the same RST result set.'
(A)
*VWRITE,'NMODES=',NMODES,', SENSOR_0=',NS0,', SENSOR_120=',NS120,', SENSOR_240=',NS240
(A,I8,A,I8,A,I8,A,I8)
*CFCLOS

ALLSEL,ALL
FINISH
